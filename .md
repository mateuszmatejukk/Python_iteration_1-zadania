class Order:

    def __init__(self, id: int, country: str, amount: float, status: str):
        self.id = id
        self.country = country
        self.amount = amount
        self.status = status

    def __str__(self) -> str:
        return f'Order(id: {self.id}, country: {self.country}, amount: {self.amount}, status: {self.status})'

    def is_paid(self) -> bool:
        return self.status == 'paid'
